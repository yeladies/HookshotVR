cmake_minimum_required(VERSION 3.20)
project(hookshot_xr_smoke LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "This isolated experiment requires Windows x64")
endif()
find_path(OPENXR_INCLUDE_DIR openxr/openxr.h REQUIRED)
find_path(VULKAN_INCLUDE_DIR vulkan/vulkan.h REQUIRED)
add_executable(hookshot_xr_smoke xr_smoke.cpp)
target_compile_features(hookshot_xr_smoke PRIVATE cxx_std_17)
target_include_directories(hookshot_xr_smoke PRIVATE "${OPENXR_INCLUDE_DIR}" "${VULKAN_INCLUDE_DIR}")
if(MSVC)
  target_compile_features(hookshot_xr_smoke PRIVATE cxx_std_20)
  target_compile_options(hookshot_xr_smoke PRIVATE /W4 /EHsc)
  target_sources(hookshot_xr_smoke PRIVATE window_capture.cpp)
  target_compile_definitions(hookshot_xr_smoke PRIVATE HOOKSHOT_WINDOW_CAPTURE=1)
  target_link_libraries(hookshot_xr_smoke PRIVATE windowsapp d3d11 dxgi)
else()
  target_compile_options(hookshot_xr_smoke PRIVATE -Wall -Wextra -Wno-missing-field-initializers)
  target_link_options(hookshot_xr_smoke PRIVATE -static)
endif()
enable_testing()
add_test(NAME xr_smoke_help COMMAND hookshot_xr_smoke --help)
add_test(NAME xr_smoke_reject_relative_loader COMMAND hookshot_xr_smoke --loader relative.dll)
set_tests_properties(xr_smoke_reject_relative_loader PROPERTIES WILL_FAIL TRUE)
add_executable(capture_guard_test capture_guard_test.cpp)
target_compile_features(capture_guard_test PRIVATE cxx_std_17)
add_test(NAME capture_guard_process_lifecycle COMMAND capture_guard_test)
set_tests_properties(capture_guard_process_lifecycle PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 30)
add_executable(panel_color_test panel_color_test.cpp)
target_compile_features(panel_color_test PRIVATE cxx_std_17)
target_include_directories(panel_color_test PRIVATE "${VULKAN_INCLUDE_DIR}")
add_test(NAME panel_color_regression COMMAND panel_color_test)
