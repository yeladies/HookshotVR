cmake_minimum_required(VERSION 3.24)
project(HookshotVRBootstrap VERSION 0.1.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(OPENXR_SDK_SOURCE_DIR "" CACHE PATH "Local Khronos OpenXR-SDK checkout")
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(BUILD_CONFORMANCE_TESTS OFF CACHE BOOL "" FORCE)
set(BUILD_API_LAYERS OFF CACHE BOOL "" FORCE)
set(BUILD_SDK_TESTS OFF CACHE BOOL "" FORCE)
set(DYNAMIC_LOADER OFF CACHE BOOL "" FORCE)
if(OPENXR_SDK_SOURCE_DIR)
    add_subdirectory("${OPENXR_SDK_SOURCE_DIR}" openxr-sdk EXCLUDE_FROM_ALL)
else()
    include(FetchContent)
    FetchContent_Declare(openxr_sdk
        GIT_REPOSITORY https://github.com/KhronosGroup/OpenXR-SDK.git
        GIT_TAG 24a1f2b92d212501a2b1b8d30ffd53abf55fcfa8)
    FetchContent_MakeAvailable(openxr_sdk)
endif()

add_library(hookshot_vr_probe_core STATIC src/probe.cpp)
target_include_directories(hookshot_vr_probe_core PUBLIC src)
target_link_libraries(hookshot_vr_probe_core PUBLIC OpenXR::headers)
if(MSVC)
    target_compile_options(hookshot_vr_probe_core PRIVATE /W4)
else()
    target_compile_options(hookshot_vr_probe_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_executable(hookshot_vr_probe src/probe_main.cpp)
target_link_libraries(hookshot_vr_probe PRIVATE hookshot_vr_probe_core OpenXR::openxr_loader)
add_executable(hookshot_vr_probe_tests tests/probe_test.cpp)
target_link_libraries(hookshot_vr_probe_tests PRIVATE hookshot_vr_probe_core)
set(VULKAN_HEADERS_SOURCE_DIR "" CACHE PATH "Optional local Vulkan-Headers checkout")
if(VULKAN_HEADERS_SOURCE_DIR)
    set(vr_vulkan_include "${VULKAN_HEADERS_SOURCE_DIR}/include")
else()
    include(FetchContent)
    FetchContent_Declare(vr_vulkan_headers
        GIT_REPOSITORY https://github.com/KhronosGroup/Vulkan-Headers.git
        GIT_TAG 577baa05033cf1d9236b3d078ca4b3269ed87a2b)
    FetchContent_MakeAvailable(vr_vulkan_headers)
    set(vr_vulkan_include "${vr_vulkan_headers_SOURCE_DIR}/include")
endif()
add_executable(hookshot_vr_scene src/session_main.cpp)
target_include_directories(hookshot_vr_scene PRIVATE "${vr_vulkan_include}" src)
target_link_libraries(hookshot_vr_scene PRIVATE OpenXR::openxr_loader ${CMAKE_DL_LIBS})
add_executable(hookshot_vr_scene_tests tests/scene_math_test.cpp)
target_include_directories(hookshot_vr_scene_tests PRIVATE src)
target_link_libraries(hookshot_vr_scene_tests PRIVATE OpenXR::headers)
if(MINGW)
    # Run from Explorer or a plain terminal without adding a compiler's DLLs to PATH.
    target_link_options(hookshot_vr_probe PRIVATE -static)
    target_link_options(hookshot_vr_probe_tests PRIVATE -static)
    target_link_options(hookshot_vr_scene PRIVATE -static)
    target_link_options(hookshot_vr_scene_tests PRIVATE -static)
endif()
enable_testing()
add_test(NAME vr_probe_contract COMMAND hookshot_vr_probe_tests)
add_test(NAME vr_scene_math COMMAND hookshot_vr_scene_tests)
